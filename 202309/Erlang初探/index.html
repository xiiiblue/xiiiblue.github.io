# 安装erlang版本25
brew install erlang@25
# rebar3下载
wget https://s3.amazonaws.com/rebar3/rebar3 && chmod +x rebar3
# 新建一个简单app项目
$ rebar3 new app <projName>
# 新建一个多app项目
$ rebar3 new release <projName>
# 新建一个简单app项目
$ rebar3 new lib <projName>
# 新建一个rebar插件
$ rebar3 new plugin <projName>
