简易配置

vi mysql_sync.json

{
  "job": {
    "setting": {
      "speed": {
        "channel": 1
      }
    },
    "content": [
      {
        "reader": {
          "name": "mysqlreader",
          "parameter": {
            "username": "root",
            "password": "password",
            "connection": [
              {
                "querySql": ["select * from info_site where id < 300;"],
                "jdbcUrl": ["jdbc:mysql://10.194.99.2:3306/hydrabot_info_prd"]
              }
            ]
          }
        },
        "writer": {
          "name": "mysqlwriter",
          "parameter": {
            "writeMode": "insert",
            "username": "root",
            "password": "password",
            "column": ["*"],
            "session": ["set session sql_mode='ANSI'"],
            "preSql": ["delete from test"],
            "connection": [
              {
                "jdbcUrl": "jdbc:mysql://10.194.98.8:3306/hydrabot_info_prd?useUnicode=true&characterEncoding=gbk",
                "table": ["test"]
              }
            ]
          }
        }
      }
    ]
  }
}
