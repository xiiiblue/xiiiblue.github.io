KubeCM常用操作

发布于2023-05-15阅读
Kubernetes

官网

https://github.com/sunny0826/kubecm

安装

# macOS下brew安装
brew install kubecm

# Linux下二进制安装
wget https://github.com/sunny0826/kubecm/releases/download/v0.23.0/kubecm_v0.23.0_Linux_x86_64.tar.gz

常用操作

# 配置Alias（可选）
alias k=kubectl
alias kc=kubecm

# 添加一个集群至～/.kube/config
kubecm add -f config.qke

# 重命名
kubecm rename

# 移除集群
kubecm delete

# 查看集群
kubecm list
kc l

# 切换集群
kubecm switch
kc s
